cmake_minimum_required(VERSION 3.16)
project(g1_rviz_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(g1_camera_angle_panel SHARED src/camera_angle_panel.cpp include/g1_rviz_plugins/camera_angle_panel.hpp)
target_include_directories(g1_camera_angle_panel PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(g1_camera_angle_panel rviz_common rclcpp std_msgs pluginlib)
target_link_libraries(g1_camera_angle_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS g1_camera_angle_panel LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  enable_testing()
  add_executable(test_camera_angle_panel test/test_camera_angle_panel.cpp)
  target_include_directories(test_camera_angle_panel PRIVATE include)
  target_link_libraries(test_camera_angle_panel g1_camera_angle_panel Qt5::Test Qt5::Widgets)
  ament_target_dependencies(test_camera_angle_panel rviz_common rclcpp std_msgs pluginlib)
  add_test(NAME camera_angle_panel COMMAND test_camera_angle_panel)
  set_tests_properties(camera_angle_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;ROS_DOMAIN_ID=87;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST")
endif()
ament_package()
