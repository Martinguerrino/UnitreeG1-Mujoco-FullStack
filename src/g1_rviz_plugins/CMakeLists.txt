cmake_minimum_required(VERSION 3.16)
project(g1_rviz_plugins)
find_package(ament_cmake REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
add_library(g1_camera_panel SHARED src/camera_panel.cpp include/g1_rviz_plugins/camera_panel.hpp
  src/perception_panel.cpp include/g1_rviz_plugins/perception_panel.hpp)
target_compile_features(g1_camera_panel PUBLIC cxx_std_17)
target_include_directories(g1_camera_panel PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(g1_camera_panel pluginlib rclcpp rviz_common std_msgs)
target_link_libraries(g1_camera_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS g1_camera_panel EXPORT camera_panel_targets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(camera_panel_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(pluginlib rclcpp rviz_common std_msgs)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_camera_panel test/test_camera_panel.cpp test/test_perception_panel.cpp)
  target_link_libraries(test_camera_panel g1_camera_panel Qt5::Widgets)
  set_tests_properties(test_camera_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
ament_package()
